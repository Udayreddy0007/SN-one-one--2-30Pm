---> Variables : a variable is a container which stores different types of data under a name / Variable Name
  
    -- Variable Name : the name  which is defined by user to store the data of different data types

        ex : 
                  employeeName = "Swathi";
   
---> Rules of Variable Name  : 
    
      
       -- A variable  name can  starts with a _(underscore)  or $ (doller)(except this symbol($) we wont consider  other special symbols)
       -- A variable name cannot starts with numbers
         
         ex : 1emp = "swathi"; -- we dont consider this 
       
       -- A variable name may consists of numbers
            
            ex : em1 ="uday";
                 e_m1 ="uday";
                 e$m1="uday";
                 1_em$1 ="uday";
       -- default keywords cannot be considered as variable names
          
          ex : for , while , count , if

     (=+ , =- , =/ , =* , =%)
